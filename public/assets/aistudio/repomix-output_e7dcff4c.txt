using namespace hv;

// 检测凝思系统中指定进程名是否存在
static bool IsProcessRunning(const char *processName)
{
    char cmd[128];
    // 同时兼容 pgrep 和 pidof 命令
    snprintf(cmd, sizeof(cmd), "pgrep -f %s > /dev/null 2>&1 || pidof %s > /dev/null 2>&1", processName, processName);
    return (system(cmd) == 0);
}

int main(int argc, char *argv[])
{
    // 1. 安全不写日志文件
    hlog_set_file("/dev/null");

    // 2. 监测 svc_scada_proxy_c 进程是否就绪
    const char *target_process = "svc_scada_proxy_c";
    std::cout << "[Monitor] Checking for process: " << target_process << " ..." << std::endl;

    while (!IsProcessRunning(target_process))
    {
        std::cout << "[Monitor] " << target_process << " not running yet. Retrying in 1s..." << std::endl;
        sleep(1);
    }

    std::cout << "[Monitor] \033[32mProcess " << target_process << " is running!\033[0m" << std::endl;
    std::cout << "[Monitor] Waiting 3 seconds for SCADA system to stabilize..." << std::endl;
    sleep(3); // 进程起来后再稳定等待 3 秒

    // 3. 初始化 SCADA 共享内存应用
    if (!scada::Init())
    {
        std::cerr << "[Error] Failed to initialize SCADA System!" << std::endl;
        return -1;
    }
    std::cout << "[Info] SCADA System initialized successfully." << std::endl;

    // ★★★ 启动后台 DNET 实时告警监听线程 ★★★
    scada::StartWarnListener();
    std::cout << "[Info] SCADA Realtime Warn Listener started." << std::endl;

    // 4. 注册路由并启动 HTTP 服务
    HttpService router;
    controller::RegisterScadaRoutes(router);

    HttpServer server;
    server.registerHttpService(&router);
    server.setPort(36581);
    server.setThreadNum(1); // 单线程稳定运行

    std::cout << "[Info] SCADA HTTP Server is listening on http://0.0.0.0:36581" << std::endl;

    // 5. 阻塞运行
    server.run();

    // 6. 退出清理
    scada::StopWarnListener();
    scada::Cleanup();
    return 0;
}

    static std::string gbk_to_utf8(const char *src)
    {
        if (!src || src[0] == '\0')
            return std::string();
        iconv_t cd = iconv_open("UTF-8//IGNORE", "GB18030");
        if (cd == (iconv_t)-1)
        {
            cd = iconv_open("UTF-8//IGNORE", "GBK");
            if (cd == (iconv_t)-1)
                return std::string(src);
        }
        size_t inbytesleft = strlen(src);
        size_t outbytesleft = inbytesleft * 4 + 1;
        std::string out;
        out.resize(outbytesleft);
        char *inbuf = const_cast<char *>(src);
        char *outbuf = &out[0];
        char *outbuf_start = outbuf;
        if (iconv(cd, &inbuf, &inbytesleft, &outbuf, &outbytesleft) == (size_t)-1)
        {
            iconv_close(cd);
            return std::string(src);
        }
        iconv_close(cd);
        out.resize(outbuf - outbuf_start);
        return out;
    }

    static std::string json_escape(const std::string &s)
    {
        std::string out;
        out.reserve(s.size() + 8);
        for (size_t i = 0; i < s.size(); ++i)
        {
            unsigned char c = (unsigned char)s[i];
            switch (c)
            {
            case '"':
                out += "\\\"";
                break;
            case '\\':
                out += "\\\\";
                break;
            case '\b':
                out += "\\b";
                break;
            case '\f':
                out += "\\f";
                break;
            case '\n':
                out += "\\n";
                break;
            case '\r':
                out += "\\r";
                break;
            case '\t':
                out += "\\t";
                break;
            default:
                if (c < 0x20)
                {
                    char buf[8];
                    sprintf(buf, "\\u%04x", c);
                    out += buf;
                }
                else
                    out += (char)c;
            }
        }
        return out;
    }

    class JsonWriter
    {
    public:
        JsonWriter(bool pretty = false, int indent = 2)
            : m_pretty(pretty), m_indent(indent), m_level(0), m_firstInContainer(true), m_afterKey(false) {}
        void beginObject()
        {
            preValue();
            m_ss += '{';
            m_level++;
            m_firstInContainer = true;
        }
        void endObject()
        {
            m_level--;
            newlineIndent();
            m_ss += '}';
            m_firstInContainer = false;
        }
        void beginArray()
        {
            preValue();
            m_ss += '[';
            m_level++;
            m_firstInContainer = true;
        }
        void endArray()
        {
            m_level--;
            newlineIndent();
            m_ss += ']';
            m_firstInContainer = false;
        }
        void key(const std::string &k)
        {
            if (m_firstInContainer)
                m_firstInContainer = false;
            else
            {
                m_ss += ',';
                newlineIndent();
            }
            m_ss += '"';
            m_ss += json_escape(k);
            m_ss += "\":";
            if (m_pretty)
                m_ss += ' ';
            m_afterKey = true;
        }
        void valueString(const std::string &v)
        {
            preValue();
            m_ss += '"';
            m_ss += json_escape(v);
            m_ss += '"';
        }
        void valueInt(int v)
        {
            preValue();
            char buf[32];
            sprintf(buf, "%d", v);
            m_ss += buf;
        }
        void valueInt(long long v)
        {
            preValue();
            char buf[32];
            sprintf(buf, "%lld", v);
            m_ss += buf;
        }
        void valueDouble(double v)
        {
            preValue();
            char buf[64];
            if (v == (long long)v && v < 1e15 && v > -1e15)
                sprintf(buf, "%.0f", v);
            else
                sprintf(buf, "%.4f", v);
            m_ss += buf;
        }
        void valueBool(bool v)
        {
            preValue();
            m_ss += v ? "true" : "false";
        }
        void rawKV(const std::string &k, const std::string &v)
        {
            key(k);
            valueString(v);
        }
        void rawKV(const std::string &k, const char *v)
        {
            key(k);
            valueString(v ? v : "");
        }
        void rawKV(const std::string &k, int v)
        {
            key(k);
            valueInt(v);
        }
        void rawKV(const std::string &k, long long v)
        {
            key(k);
            valueInt(v);
        }
        void rawKV(const std::string &k, double v)
        {
            key(k);
            valueDouble(v);
        }
        void rawKV(const std::string &k, bool v)
        {
            key(k);
            valueBool(v);
        }
        std::string str() const { return m_ss; }

    private:
        void preValue()
        {
            if (m_afterKey)
                m_afterKey = false;
            else
            {
                if (m_firstInContainer)
                    m_firstInContainer = false;
                else
                {
                    m_ss += ',';
                    newlineIndent();
                }
            }
        }
        void newlineIndent()
        {
            if (!m_pretty)
                return;
            m_ss += '\n';
            m_ss.append(m_level * m_indent, ' ');
        }
        bool m_pretty;
        int m_indent;
        int m_level;
        std::string m_ss;
        bool m_firstInContainer;
        bool m_afterKey;
    };

    template <typename T>
    static std::vector<T> fetch_table(short app_no, int table_no)
    {
        std::vector<T> v;
        TableOp tab;

        // 1. 尝试在指定域打开 (如 SCADA)
        short open_app = app_no;
        if (tab.TableOpen(open_app, table_no) < 0)
        {
            // 2. 失败则尝试在 PUBLIC 域打开
            open_app = (app_no == SCADA) ? PUBLIC : SCADA;
            if (tab.TableOpen(open_app, table_no) < 0)
            {
                return v;
            }
        }

        short field_num = 0;
        int rec_num = 0, rec_size = 0;
        // 获取该表真实的记录数和每条大小
        if (tab.GetTablePara(field_num, rec_num, rec_size) >= 0 && rec_num > 0)
        {
            int total_bytes = rec_num * (int)sizeof(T);
            char *buf = (char *)malloc(total_bytes);
            if (buf)
            {
                int get_num = tab.TableGet(buf, total_bytes);
                if (get_num > 0)
                {
                    v.resize(get_num);
                    memcpy(v.data(), buf, (size_t)get_num * sizeof(T));
                }
                free(buf);
            }
        }
        tab.TableClose();

        // 3. 如果 TableOp 读不到，最后用大缓冲区的 GetAllRec 兜底
        if (v.empty())
        {
            int bufSize = 512 * 1024; // 直接给 512KB 大缓冲
            char *buf = (char *)malloc(bufSize);
            if (buf)
            {
                int n = GetAllRec(app_no, table_no, buf, bufSize);
                if (n <= 0)
                    n = GetAllRec(PUBLIC, table_no, buf, bufSize);
                if (n > 0)
                {
                    v.resize(n);
                    memcpy(v.data(), buf, (size_t)n * sizeof(T));
                }
                free(buf);
            }
        }

        return v;
    }

    // ---------------- DNET 发送底层（单次按需发送） ----------------
    static int SendYkYtByDnet(system_net_info *sysnet, const char *hostname,
                              int report_type, int dev_id, int yk_id,
                              int up_or_down, unsigned char yk_state,
                              unsigned char channel_no,
                              float set_value, bool is_yt)
    {
        if (!sysnet)
            return -1;
        int tmp_i = 0;
        char tmp_c = 0;
        int number = 1;
        int report_len = is_yt ? 21 : 17;

        CPackStream PackBuf;
        PackBuf.OpenForWrite();
        PackBuf << report_type << tmp_i << tmp_i << tmp_i << tmp_i;
        PackBuf << number << report_len << tmp_c << tmp_c << tmp_c << tmp_c;
        PackBuf << dev_id << yk_id << up_or_down << yk_state << channel_no;
        if (is_yt)
        {
            PackBuf << set_value;
        }

        char *buffer = NULL;
        int counter = 0;
        PackBuf.GetWriteBuffer(buffer, counter);

        int ret = sysnet->dnet_send_message(
            DNET_REPORT_POINT_TO_POINT,
            report_type,
            (char *)hostname,
            (char *)"FRONT_NET",
            buffer, counter);

        free(buffer);
        return ret;
    }

    static int SendOperateYkYt(system_net_info *sysnet, const char *hostname,
                               bool is_yt, int dev_id, int yk_id, int status,
                               int yk_mode, int up_or_down,
                               float old_value, float new_value,
                               const char *oper_name)
    {
        if (!sysnet)
            return -1;
        int report_type = is_yt ? DNET_OPERATE_YT : DNET_OPERATE_YK;
        time_t tmptime;
        time(&tmptime);
        int int_time = (int)tmptime;
        short value_syb_field_no = -1;

        CPackStream PackBuf;
        PackBuf.OpenForWrite();
        PackBuf << report_type << int_time << dev_id << yk_id << value_syb_field_no;
        PackBuf << status << yk_mode << up_or_down;
        if (is_yt)
        {
            PackBuf << old_value;
            PackBuf << new_value;
        }
        PackBuf.set_char((char *)hostname, 32);
        PackBuf.set_char((char *)oper_name, 32);

        char *buffer = NULL;
        int counter = 0;
        PackBuf.GetWriteBuffer(buffer, counter);

        int ret = sysnet->dnet_send_message(
            DNET_REPORT_BROADCAST_SAFE,
            report_type,
            DNET_ALL_HOST,
            (char *)"svc_scada_proxy",
            buffer, counter);

        free(buffer);
        return ret;
    }

        void BuildIndex()
        {
            m_fac2bays.clear();
            for (auto &b : m_bays)
            {
                // 1. 优先通过 CDbMap 反查
                int facId = m_dbmap.GetFacIdxByIdx(b.bay_id);

                // 2. ★★★ 终极容错：如果 CDbMap 返回 -1，直接使用全局唯一的单厂站兜底 ★★★
                if (facId <= 0 && m_facs.size() == 1)
                {
                    facId = m_facs[0].fac_id;
                }

                // 3. 如果依然 <= 0，尝试从别名前缀智能归属 (如 F01.xxx 归入 F01)
                if (facId <= 0)
                {
                    for (auto &f : m_facs)
                    {
                        if (strncmp(b.bay_alias, f.fac_alias, strlen(f.fac_alias)) == 0)
                        {
                            facId = f.fac_id;
                            break;
                        }
                    }
                }

                if (facId > 0)
                    m_fac2bays[facId].push_back(b.bay_id);
            }

            for (auto &d : m_cbs)
            {
                int bay = m_dbmap.GetBayIdxByIdx(d.cb_id);
                // 容错：如果反查失败，按别名前缀或者单间隔兜底
                if (bay <= 0 && m_bays.size() == 1)
                {
                    bay = m_bays[0].bay_id;
                }
                if (bay <= 0)
                {
                    for (auto &b : m_bays)
                    {
                        if (strncmp(d.cb_alias, b.bay_alias, strlen(b.bay_alias)) == 0)
                        {
                            bay = b.bay_id;
                            break;
                        }
                    }
                }
                if (bay > 0)
                    m_bay2cbs[bay].push_back(d.cb_id);
            }

            for (auto &d : m_bss)
            {
                int bay = m_dbmap.GetBayIdxByIdx(d.bs_id);
                if (bay > 0)
                    m_bay2bss[bay].push_back(d.bs_id);
            }
            for (auto &d : m_bls)
            {
                int bay = m_dbmap.GetBayIdxByIdx(d.bl_id);
                if (bay > 0)
                    m_bay2bls[bay].push_back(d.bl_id);
            }
            for (auto &d : m_trs)
            {
                int bay = m_dbmap.GetBayIdxByIdx(d.tr_id);
                if (bay > 0)
                    m_bay2trs[bay].push_back(d.tr_id);
            }
            for (auto &d : m_gns)

                if (bay > 0)
                    m_bay2gns[bay].push_back(d.gn_id);
            }
            for (auto &d : m_lds)

                int bay = m_dbmap.GetBayIdxByIdx(d.ld_id);
                if (bay > 0)
                    m_bay2lds[bay].push_back(d.ld_id);
            }
            for (auto &d : m_rcs)

        bool GetYkPointInfoById(int yk_id, YkPointInfo &out)
        {
            yk_yx_rely_scada yk;
            if (LdbGetRecordById(YK_YX_RELY_NO, (char *)&yk_id, (char *)&yk, sizeof(yk)) <= 0)
                return false;
            out.yk_id = yk.yk_id;
            out.yk_type = yk.yk_type;
            out.channel = yk.current_channel;
            out.valid = true;
            strncpy(out.yk_name, yk.yk_name, sizeof(out.yk_name) - 1);
            strncpy(out.yk_alias, yk.yk_alias, sizeof(out.yk_alias) - 1);

            for (auto &c : m_ykChannels)

                out.exec_type = c.exec_type;

            out.dev_id = m_dbmap.GetDevIdxByIdx(yk_id);

        static bool SendReportToFront_Native(system_net_info *sysnet, int fore_machine_type, int report_type, char *buffer, int counter)
        {
            char hostname[32] = {0};
            ns_gethostname(hostname, 32);

            int size = 8;
            char *tmpstr = (char *)malloc(size);
            int recNum = GetAllRec(SCADA, MACHINE_INFO_NO, tmpstr, size);
            if (recNum <= 0)
            {
                if (tmpstr)
                    free(tmpstr);
                return false;
            }

            int system_type = -1, machine_type = -1;
            machine_info_scada *pmachine_info = NULL;

            for (int i = 0; i < recNum; i++)
            {
                pmachine_info = (machine_info_scada *)(tmpstr + sizeof(machine_info_scada) * i);
                if (strcmp(pmachine_info->machine_name, hostname) != 0)
                    continue;
                system_type = pmachine_info->system_type;
                machine_type = pmachine_info->machine_type;
                break;
            }

            bool ret = false;
            for (int i = 0; i < recNum; i++)
            {
                pmachine_info = (machine_info_scada *)(tmpstr + sizeof(machine_info_scada) * i);
                if (system_type == 3) // 调试系统
                {
                    if ((machine_type != pmachine_info->machine_type) || (pmachine_info->system_type == 3))
                        continue;
                }
                else
                {
                    if (pmachine_info->system_type != system_type)
                        continue;
                }

                int tt = (0x01 << (pmachine_info->machine_type - 1));
                if ((tt & fore_machine_type) != 0 || fore_machine_type == -1)
                {
                    int retval = sysnet->dnet_send_message(
                        DNET_REPORT_POINT_TO_POINT,
                        report_type,
                        pmachine_info->machine_name,
                        (char *)"FRONT_NET",
                        buffer, counter);

                    std::cout << "  -> [前置路由] 投递到前置节点: " << pmachine_info->machine_name
                              << ", 报文类型=" << report_type << ", 结果=" << retval << std::endl;
                    if (retval >= 0)
                        ret = true;
                }
            }
            free(tmpstr);

            // 兜底发给本机前置
            if (!ret)
            {
                int retval = sysnet->dnet_send_message(
                    DNET_REPORT_POINT_TO_POINT,
                    report_type,
                    hostname,
                    (char *)"FRONT_NET",
                    buffer, counter);
                std::cout << "  -> [前置路由] 本机 FRONT_NET 兜底发送结果=" << retval << std::endl;
                if (retval >= 0)
                    ret = true;
            }
            return ret;
        }

        // [步骤 2] 原版 CYkDialog.cpp::SendReport 纯正封包实现 (17字节严格报文)
        // =========================================================================
        static bool NativeSendYkReport(int dev_id, int yk_id, int channel_no, unsigned char yk_state, int report_type)
        {
            system_net_info sysnet((char *)"scada_dump", DNET_NO);
            CDbMap dbmap;

            // 1. 查询关联 IED 得到前置机号 (原版第 637~646 行)
            int fore_machine_type = -1;
            COrderMap ordermap;
            yk_channel_scada yk_channel;
            if (ordermap.GetYkChannel(yk_id, channel_no, &yk_channel) >= 0)
            {
                int ied_idx = dbmap.GetIedIdxByAlias(yk_channel.ld_alias);
                if (ied_idx > 0)
                {
                    ied_info_scada ied_info;
                    if (LdbGetRecordById(IED_INFO_NO, (char *)&ied_idx, (char *)&ied_info, sizeof(ied_info)) > 0)
                    {
                        fore_machine_type = ied_info.front_no | ied_info.link_node;
                    }
                }
            }

            // 2. 封包 (原版 CYkDialog.cpp 第 618~636 行，严格按原版类型推入 CPackStream)
            int number = 1;
            int report_len = 17;
            int tmp_i = 0;
            char tmp_c = 0;
            int up_or_down = 0; // 原版第 339 行明确规定：开关遥控 up_or_down 恒为 0！

            unsigned char send_yk_state = yk_state;
            unsigned char send_channel = (unsigned char)channel_no;

            CPackStream PackBuf;
            PackBuf.OpenForWrite();
            PackBuf << report_type;
            PackBuf << tmp_i;
            PackBuf << tmp_i;
            PackBuf << tmp_i;
            PackBuf << tmp_i;
            PackBuf << number;
            PackBuf << report_len;
            PackBuf << tmp_c;
            PackBuf << tmp_c;
            PackBuf << tmp_c;
            PackBuf << tmp_c;
            PackBuf << dev_id;
            PackBuf << yk_id;
            PackBuf << up_or_down;
            PackBuf << send_yk_state; // 严格以 unsigned char 压入 1 字节
            PackBuf << send_channel;  // 严格以 unsigned char 压入 1 字节

            char *send_buffer = NULL;
            int counter = 0;
            PackBuf.GetWriteBuffer(send_buffer, counter);

            std::cout << "[Step] 发送前置报文: report_type=" << report_type
                      << ", 设备ID=" << dev_id
                      << ", 遥控点ID=" << yk_id
                      << ", 状态码(yk_state)=" << (int)send_yk_state
                      << ", 通道号=" << (int)send_channel
                      << ", 报文大小=" << counter << " 字节" << std::endl;

            bool send_ok = SendReportToFront_Native(&sysnet, fore_machine_type, report_type, send_buffer, counter);
            free(send_buffer);

            // 3. 记录操作审计流水 (原版 CYkDialog.cpp 第 668~685 行)
            if (send_ok)
            {
                char hostname[32] = {0};
                ns_gethostname(hostname, 32);
                int oper_status = (report_type == DNET_YK_PRE_SET) ? 1 : 4;
                time_t now = time(NULL);

                CPackStream OperPack;
                OperPack.OpenForWrite();
                int rtype = DNET_OPERATE_YK;
                short fno = -1;
                OperPack << rtype << (int)now << dev_id << yk_id;
                OperPack << fno;
                OperPack << oper_status << (int)send_yk_state << up_or_down;
                OperPack.set_char(hostname, 32);
                OperPack.set_char((char *)"test_operator", 32);

                char *obuf = NULL;
                int olen = 0;
                OperPack.GetWriteBuffer(obuf, olen);
                sysnet.dnet_send_message(DNET_REPORT_BROADCAST_SAFE, DNET_OPERATE_YK, DNET_ALL_HOST, (char *)"svc_scada_proxy", obuf, olen);
                free(obuf);
            }

            return send_ok;
        }

        int DoYk(int yk_id, const char *action, int yk_state)
        {
            // 1. 从共享内存点表读取该遥控点详情
            yk_yx_rely_scada yk_info;
            if (LdbGetRecordById(YK_YX_RELY_NO, (char *)&yk_id, (char *)&yk_info, sizeof(yk_info)) <= 0)
                return -1;

            // 2. ★ 后端自动根据 yk_id 反查出它归属的开关设备 ID (完全不需要前端传！)
            CDbMap dbmap;
            int dev_id = dbmap.GetDevIdxByIdx(yk_id);
            if (dev_id <= 0)
                return -1;

            // 3. 通道号也由后端从 yk_info.current_channel 自动提取
            unsigned char send_channel = (unsigned char)yk_info.current_channel;
            unsigned char send_yk_state = (unsigned char)yk_state;

            // 1. 报文类型与状态完全对齐原版 CYkDialog
            int report_type = 0;
            if (strcmp(action, "prev") == 0)
            {
                report_type = DNET_YK_PRE_SET; // 5520
            }
            else if (strcmp(action, "exec") == 0)
            {
                report_type = DNET_YK_EXECUTE; // 5521
            }
            else if (strcmp(action, "cancel") == 0)
            {
                report_type = DNET_YK_CANCEL; // 5522
            }
            else if (strcmp(action, "direct") == 0)
            {
                report_type = DNET_YK_DIRECT; // 5532
            }
            else
            {
                return -1;
            }

            // 4. 调用刚才测试完全成功的正规 17 字节单字节封包与路由
            return NativeSendYkReport(dev_id, yk_id, send_channel, send_yk_state, report_type);
        }

        int DoYt(int yt_id, const char *action, const char *oper, float set_value, float old_value = -1.0f)
        {
            YkPointInfo info;
            if (!GetYkPointInfoById(yt_id, info))
                return -1;
            if (!is_yt_type(info.yk_type))
                return -1;
            if (info.dev_id <= 0)
                return -1;

            int report_type = 0;
            int status = 0;
            if (strcmp(action, "prev") == 0)
            {
                report_type = DNET_YT_PRE_SET;
                status = 1;
            }
            else if (strcmp(action, "exec") == 0)
            {
                report_type = DNET_YT_EXECUTE;
                status = 4;
            }
            else if (strcmp(action, "cancel") == 0)
            {
                report_type = DNET_YT_CANCEL;
                status = 5;
            }
            else if (strcmp(action, "direct") == 0)
            {
                report_type = DNET_YT_DIRECT;
                status = 4;
            }
            else
                return -1;

            int up_or_down = 0;
            unsigned char yk_state = 0;
            if (strcmp(oper, "up") == 0)
            {
                up_or_down = 1;
                yk_state = YT_STATE_UP;
            }
            else if (strcmp(oper, "down") == 0)
            {
                up_or_down = 2;
                yk_state = YT_STATE_DOWN;
            }
            else if (strcmp(oper, "stop") == 0)
            {
                up_or_down = 3;
                yk_state = YT_STATE_STOP;
            }
            else if (strcmp(oper, "adjust") == 0)
            {
                up_or_down = 0;
                yk_state = YT_STATE_ADJUST;
            }
            else
                return -1;

            if (old_value < 0.0f)
                old_value = 0.0f;

            char local_host[64] = {0};
            ns_gethostname(local_host, sizeof(local_host));
            system_net_info sysnet((char *)"scada_dump", DNET_NO);

            int ret = SendYkYtByDnet(&sysnet, local_host, report_type, info.dev_id, yt_id, up_or_down, yk_state,
                                     (unsigned char)info.channel, set_value, true);
            if (ret < 0)
                return ret;

            SendOperateYkYt(&sysnet, local_host, true, info.dev_id, yt_id, status, yk_state, up_or_down, old_value, set_value, "scada_dump");
            return ret;
        }

        std::vector<YkFindItem> FindYkByName(const std::string &keyword, bool yt_only)
        {

            return res;
        }

        static const T *FindByIdGeneric(const std::vector<T> &v, F fn, int id)

                    return &x;
            return nullptr;
        }

    // ★★★ 极速实时数据直读接口 (使用官方轻量哈希直读 LdbGetRecordById) ★★★
    // =============================================================
    RealtimeResult QueryRealtime(const std::vector<int> &yc_ids,
                                 const std::vector<int> &yx_ids,
                                 const std::vector<int> &dd_ids)
    {
        RealtimeResult res;
        res.yc_list.reserve(yc_ids.size());
        res.yx_list.reserve(yx_ids.size());
        res.dd_list.reserve(dd_ids.size());

        // 1. 遥测直读 (微秒级直查共享内存)
        for (int ycId : yc_ids)
        {
            yc_define_scada y;
            if (LdbGetRecordById(YC_DEFINE_NO, (char *)&ycId, (char *)&y, sizeof(y)) > 0)
            {
                res.yc_list.push_back({ycId, (double)y.yc_val, (int)y.status});
            }
        }

        // 2. 遥信直读 (微秒级直查共享内存)
        for (int yxId : yx_ids)
        {
            yx_define_scada y;
            if (LdbGetRecordById(YX_DEFINE_NO, (char *)&yxId, (char *)&y, sizeof(y)) > 0)
            {
                res.yx_list.push_back({yxId, (int)y.yx_value, (int)y.status, (int)y.q});
            }
        }

        // 3. 电度直读 (微秒级直查共享内存，直接读取 kwh_2)
        for (int ddId : dd_ids)
        {
            dd_define_scada d;
            if (LdbGetRecordById(DD_DEFINE_NO, (char *)&ddId, (char *)&d, sizeof(d)) > 0)
            {
                res.dd_list.push_back({ddId, (double)d.kwh_2, (int)d.status});
            }
        }

        return res;
    }

    // =============================================================
    // ★★★ 遥信清闪：通过官方实时库接口直接写入 Linux 共享内存 ★★★
    // =============================================================
    UnflashResult ClearYxFlash(const std::vector<int> &yx_ids, bool all)
    {
        int cleared_count = 0;
        CDbMap dbmap;

        // 打开遥信表与一次设备表（整个清闪过程只打开一次，开销极小）
        TableOp yx_table, cb_table;
        if (yx_table.TableOpen(SCADA, YX_DEFINE_NO) < 0)
        {
            return {false, 0, "Failed to open YX_DEFINE table"};
        }
        cb_table.TableOpen(SCADA, CB_DEVICE_NO);

        // 核心清闪逻辑：修改后直接通过 TableUpdate 刷入共享内存
        auto do_clear_one = [&](int id)
        {
            yx_define_scada y;
            // 1. 通过官方直读接口读取当前测点
            if (LdbGetRecordById(YX_DEFINE_NO, (char *)&id, (char *)&y, sizeof(y)) > 0)
            {
                // 检查是否处于闪烁态 (7:变位合, 8:变位分, 9:事故合, 10:事故分, 11:SOE动, 12:SOE复)
                if (y.status >= STATE_YX_BW_ON && y.status <= STATE_YX_SOE_OFF)
                {
                    // ★★★ 1. 遥信测点状态复位为 0 ★★★
                    y.status = STATE_YX_NORMAL;

                    // 落实写入 Linux 共享内存实时库
                    if (yx_table.TableUpdate((char *)&y, sizeof(y)) >= 0)
                    {
                        cleared_count++;
                    }

                    // ★★★ 2. 同步将所属开关本体（cb_device）的共享内存状态复位为 0 ★★★
                    int dev_id = dbmap.GetDevIdxByIdx(id);
                    if (dev_id > 0)
                    {
                        cb_device_scada cb;
                        if (LdbGetRecordById(CB_DEVICE_NO, (char *)&dev_id, (char *)&cb, sizeof(cb)) > 0)
                        {
                            if (cb.status >= STATE_YX_BW_ON && cb.status <= STATE_YX_SOE_OFF)
                            {
                                cb.status = STATE_YX_NORMAL;                   // 开关本体也停止闪烁
                                cb_table.TableUpdate((char *)&cb, sizeof(cb)); // 写入共享内存
                            }
                        }
                    }
                }
            }
        };

        if (all)
        {
            // 【全站全部清闪】：遍历全站遥信表
            auto all_yxs = fetch_table<yx_define_scada>(SCADA, YX_DEFINE_NO);
            for (const auto &yx : all_yxs)
            {
                do_clear_one(yx.yx_id);
            }
        }
        else
        {
            // 【当前画面清闪】：精准清闪传入的测点
            for (int id : yx_ids)
            {
                do_clear_one(id);
            }
        }

        yx_table.TableClose();
        cb_table.TableClose();

        printf("[SCADA SHM] ClearYxFlash: %d points cleared in Shared Memory.\n", cleared_count);
        return {true, cleared_count, "Successfully cleared in shared memory"};
    }

    static void DoRealDnetUnflash(const std::vector<int> &key_ids)
    {
        if (key_ids.empty() || !g_pCmdSendDnet)
            return;

        char local_host[32] = {0};
        ns_gethostname(local_host, 32);

        time_t tmp_time = time(NULL);
        int int_time = (int)tmp_time;
        short field_id = 0;
        short status_field = 0;
        int count = (int)key_ids.size();
        int report_type = DNET_YX_UNFLASH;

        // 原生内存打包：4*sizeof(int) + 2*sizeof(short) + 64 + count*sizeof(int)
        int buf_len = 4 * sizeof(int) + 2 * sizeof(short) + 64 + count * sizeof(int);
        char *send_buf = (char *)malloc(buf_len);
        if (!send_buf)
            return;

        int counter = 0;
        memcpy(send_buf + counter, &report_type, sizeof(int));
        counter += sizeof(int);
        memcpy(send_buf + counter, &int_time, sizeof(int));
        counter += sizeof(int);
        memcpy(send_buf + counter, &field_id, sizeof(short));
        counter += sizeof(short);
        memcpy(send_buf + counter, &status_field, sizeof(short));
        counter += sizeof(short);
        memcpy(send_buf + counter, local_host, 32);
        counter += 32;
        memcpy(send_buf + counter, "web_operator", 32);
        counter += 32;
        memcpy(send_buf + counter, &count, sizeof(int));
        counter += sizeof(int);

        for (int id : key_ids)
        {
            memcpy(send_buf + counter, &id, sizeof(int));
            counter += sizeof(int);
        }

        // ★★★ 核心优化：使用独立写通道 g_pCmdSendDnet 发送，绝对不影响读通道！★★★
        {
            std::lock_guard<std::mutex> lk(g_dnet_send_mutex);
            // 优先发给本机前置代理服务，微秒级直接返回，绝不超时挂起
            int ret = g_pCmdSendDnet->dnet_send_message(
                DNET_REPORT_POINT_TO_POINT,
                DNET_YX_UNFLASH,
                DNET_LOCAL_HOST,
                (char *)"svc_scada_proxy",
                send_buf, counter);

            if (ret < 0)
            {
                // 兜底发给 SCADA 服务器
                g_pCmdSendDnet->dnet_send_message(
                    DNET_REPORT_POINT_TO_POINT,
                    DNET_YX_UNFLASH,
                    DNET_SERVER_HOST,
                    (char *)"svc_scada_server",
                    send_buf, counter);
            }
        }

        free(send_buf);
    }
    // 后台独立异步清闪处理线程
    static void UnflashWorker()
    {
        while (g_unflash_running.load())
        {
            std::vector<int> batch;
            {
                std::unique_lock<std::mutex> lk(g_unflash_queue_mutex);
                g_unflash_cv.wait_for(lk, std::chrono::milliseconds(50), []
                                      { return !g_unflash_point_queue.empty() || !g_unflash_running.load(); });

                if (!g_unflash_running.load())
                    break;

                while (!g_unflash_point_queue.empty())
                {
                    batch.push_back(g_unflash_point_queue.front());
                    g_unflash_point_queue.pop_front();
                }
            }

            if (!batch.empty())
            {
                // 测点全局去重，避免重复发送相同的开关/遥信 ID
                std::sort(batch.begin(), batch.end());
                batch.erase(std::unique(batch.begin(), batch.end()), batch.end());

                // 分批发送，每包最多 500 个点
                const size_t CHUNK_SIZE = 500;
                for (size_t i = 0; i < batch.size(); i += CHUNK_SIZE)
                {
                    size_t end = std::min(i + CHUNK_SIZE, batch.size());
                    std::vector<int> chunk(batch.begin() + i, batch.begin() + end);
                    DoRealDnetUnflash(chunk);
                    usleep(5 * 1000); // 微量释放
                }
            }
        }
    }

        int rec_num = GetAllRec(SCADA, WARN_TYPE_DEFINE_NO, buf, buf_size);

    static std::string GetCachedNameByIdx(CDbMap &dbmap, int id)
    {
        if (id <= 0)
            return "-";
        {
            std::lock_guard<std::mutex> lk(g_name_cache_mutex);
            auto it = g_name_cache.find(id);
            if (it != g_name_cache.end())
                return it->second;
        }

        char raw_name[128] = {0};
        dbmap.GetNameByIdx(id, raw_name, sizeof(raw_name));
        std::string utf8_name = gbk_to_utf8(raw_name);
        if (utf8_name.empty())
            utf8_name = std::to_string(id);

        {
            std::lock_guard<std::mutex> lk(g_name_cache_mutex);
            g_name_cache[id] = utf8_name;
        }
        return utf8_name;
    }

    static std::string FormatTimestamp(time_t t, int ms = 0)
    {
        struct tm tm_buf;
        localtime_r(&t, &tm_buf);
        char buf[64];
        snprintf(buf, sizeof(buf), "%04d-%02d-%02d %02d:%02d:%02d.%03d",
                 tm_buf.tm_year + 1900, tm_buf.tm_mon + 1, tm_buf.tm_mday,
                 tm_buf.tm_hour, tm_buf.tm_min, tm_buf.tm_sec, ms);
        return std::string(buf);
    }

    static void ParseFullWarnPacket(const char *buffer, int length, int byte_order)
    {
        if (!buffer || length <= 0)
            return;

        CPackStream pack;
        pack.OpenForRead((char *)buffer, length, byte_order);

        int num = 0;
        pack >> num;
        if (num <= 0 || num > 1000)
            return;

        CDbMap dbmap;
        time_t now = time(NULL);
        std::vector<WarnItem> parsed_batch;
        parsed_batch.reserve(num);

        static std::once_flag init_level_flag;
        std::call_once(init_level_flag, InitWarnLevelsFromDb);

        for (int i = 0; i < num; ++i)
        {
            WarnItem item;
            item.confirm = 0;
            item.confirm_desc = "未确认";
            item.timestamp = (uint64_t)now;
            item.time_str = FormatTimestamp(now, 0);

            pack >> item.type; // 原始类型

            // 1. 专业大类精确研判 (严格对应截图标签)
            if (item.type == 1)
                item.type_name = "遥信变位";
            else if (item.type == 2 || item.type == 22 || item.type == 23)
                item.type_name = "遥测越限";
            else if (item.type == 3)
                item.type_name = "厂站工况";
            else if (item.type == 4)
                item.type_name = "结点工况";
            else if (item.type == 8 || item.type == 9)
                item.type_name = "遥控";
            else if (item.type == 11 || item.type == 12)
                item.type_name = "人工操作";
            else if (item.type == 10)
                item.type_name = "事件SOE";
            else if (item.type == 28 || item.type == 35 || item.type == 36)
                item.type_name = "保护事件";
            else
                item.type_name = "遥信变位";

            item.my_type = (item.type >= 0 && item.type < 50) ? g_ShowArray[item.type] : 0;
            item.level = (item.type >= 0 && item.type < 50) ? g_ShowArray2[item.type] : 1;

            int warn_no = 0;
            pack >> warn_no;
            bool has_detail = ((warn_no & 0x8000) != 0 && warn_no != -1);

            int str_len = 0;
            pack >> str_len;
            if (str_len > 0 && str_len < 1024)
            {
                std::vector<char> raw_str(str_len + 1, 0);
                pack.get_char(str_len, raw_str.data());
                item.content = gbk_to_utf8(raw_str.data());
            }

            pack >> item.fac_id;
            pack >> item.key_id;
            pack >> item.field_id;
            pack >> item.value;
            item.val_desc = GetValueMeaning(item.type, item.value);

            if (has_detail)
            {
                char result_buf[65] = {0}, host_buf[33] = {0}, oper_buf[33] = {0};
                pack.get_char(64, result_buf);
                pack.get_char(32, host_buf);
                pack.get_char(32, oper_buf);
                std::string res_u = gbk_to_utf8(result_buf);
                std::string op_u = gbk_to_utf8(oper_buf);
                if (!res_u.empty())
                    item.content += " [" + res_u + "]";
                if (!op_u.empty())
                    item.content += " [操作人:" + op_u + "]";
            }

            // 2. 查库内测点属性精准研判严重级别 (原版 main_view_warn.cpp 规则)
            int table_no = item.key_id / 1000000;
            if (table_no == YX_DEFINE_NO)
            {
                yx_define_scada yx_stru;
                if (LdbGetRecordById(YX_DEFINE_NO, (char *)&item.key_id, (char *)&yx_stru, sizeof(yx_define_scada)) > 0)
                {
                    for (int k = 0; k < g_warn_level_count; ++k)
                    {
                        if (g_LevelDefines[k].warntype_id == yx_stru.int_5)
                        {
                            item.level = g_LevelDefines[k].warn_level;
                            break;
                        }
                    }
                }
                // 事故分闸/跳闸特例研判
                if (item.content.find("事故") != std::string::npos &&
                    (item.content.find("跳闸") != std::string::npos || item.content.find("分闸") != std::string::npos))
                {
                    item.level = g_sg_level;
                }
            }
            else if (table_no == YC_DEFINE_NO)
            {
                item.level = 2; // 越限默认归为告警
            }
            else if (table_no == CB_DEVICE_NO)
            {
                if (item.content.find("事故跳闸") != std::string::npos)
                {
                    item.level = g_sg_level;
                }
            }

            // 匹配级别中文与颜色
            if (item.level == 3)
            {
                item.level_name = "事故信息";
                item.level_color = "#FF0000";
            }
            else if (item.level == 2)
            {
                item.level_name = "告警信息";
                item.level_color = "#E6A23C";
            }
            else
            {
                item.level_name = "提示信息";
                item.level_color = "#67C23A";
            }

            // 3. 研判业务态：检修 与 复归状态
            item.is_examine = (item.content.find("检修") != std::string::npos);
            item.is_revert = (item.value == 0 || item.content.find("复归") != std::string::npos);

            // 4. 组装归属标签集合 (前端可直接使用 tabs.includes(当前Tab))
            item.tabs.push_back("时序信息");
            item.tabs.push_back(item.level_name); // 提示信息 / 告警信息 / 事故信息
            item.tabs.push_back(item.type_name);  // 遥信变位 / 遥测越限 / 厂站工况 ...
            if (item.is_examine)
            {
                item.tabs.push_back("检修信息");
            }
            if (!item.is_revert && !item.is_examine)
            {
                item.tabs.push_back("未复归信息");
            }

            // 5. 拼装 SCADA 原始单行全文本
            item.full_text = item.level_name + "  " + item.time_str + "  " + item.content;

            item.status_desc = GetStatusDesc(item.status, item.value);

            // 6. 翻译名称
            if (item.key_id > 0)
            {
                item.fac_id = dbmap.GetFacIdxByIdx(item.key_id);
                item.bay_id = dbmap.GetBayIdxByIdx(item.key_id);
                item.dev_id = dbmap.GetDevIdxByIdx(item.key_id);

                item.fac_name = GetCachedNameByIdx(dbmap, item.fac_id);
                item.bay_name = GetCachedNameByIdx(dbmap, item.bay_id);
                item.dev_name = GetCachedNameByIdx(dbmap, item.dev_id);
                item.key_name = GetCachedNameByIdx(dbmap, item.key_id);
            }
            else
            {
                item.fac_name = "全站系统";
                item.bay_name = "-";
                item.dev_name = "-";
                item.key_name = "-";
            }

            parsed_batch.push_back(item);
        }

        // 推入内存队列
        {
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : parsed_batch)
            {
                item.seq_id = ++g_warn_seq_counter;
                if (g_warn_queue.size() >= MAX_WARN_CACHE)
                {
                    g_warn_queue.pop_front();
                }
                g_warn_queue.push_back(item);
            }
        }
    }

    // =============================================================
    // 高吞吐后台监听线程 (固定使用白名单 "GJ_SERVER")
    // =============================================================
    // 监听工作线程：独占读通道，无任何发送行为，零互斥锁
    static void WarnListenWorker()
    {
        // 绑定专属白名单服务名 "GJ_SERVER"
        if (!g_pWarnRecvDnet)
        {
            g_pWarnRecvDnet = new system_net_info((char *)"GJ_SERVER", DNET_NO);
        }

        const int RECV_BUF_SIZE = 65536; // 64KB 大缓冲区
        char *recv_buffer = (char *)malloc(RECV_BUF_SIZE);

        while (g_warn_running.load())
        {
            app_buf_head ahead;
            int recv_count = 0;

            // ★★★ 零锁！独占专线高速抽取，不受任何确认写操作影响 ★★★
            while (true)
            {
                int ret = -1;
                if (g_pWarnRecvDnet)
                {
                    ret = g_pWarnRecvDnet->dnet_receive_message(ahead, recv_buffer);
                }

                if (ret >= 0)
                {
                    if (ahead.rhead.report_type == DNET_SEND_MESSAGE_TO_WARN)
                    {
                        ParseFullWarnPacket(recv_buffer, ahead.rhead.length, ahead.rhead.byte_order);
                    }
                    recv_count++;
                    if (recv_count > 50)
                        break;
                }
                else
                {
                    break;
                }
            }

            if (recv_count == 0)
            {
                usleep(20 * 1000); // 无报文时释放 CPU 20ms
            }
        }

        if (recv_buffer)
        {
            free(recv_buffer);
            recv_buffer = nullptr;
        }
    }

    bool StartWarnListener()
    {
        if (g_warn_running.load())
            return true;

        // 初始化独立的发送单例
        if (!g_pCmdSendDnet)
        {
            g_pCmdSendDnet = new system_net_info((char *)"WARN_CMD_CLIENT", DNET_NO);
        }

        g_warn_running.store(true);
        g_warn_thread = std::thread(WarnListenWorker);
        g_warn_thread.detach();

        g_unflash_running.store(true);
        g_unflash_thread = std::thread(UnflashWorker);
        g_unflash_thread.detach();

        return true;
    }

    void StopWarnListener()
    {
        g_warn_running.store(false);
        g_unflash_running.store(false);
        g_unflash_cv.notify_all();
    }

    // =============================================================
    // 实时告警查询接口 (彻底修复取数反向导致的停滞定格 Bug)
    // =============================================================
    std::vector<WarnItem> GetRealtimeWarns(int limit, uint64_t after_seq_id)
    {
        std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
        std::vector<WarnItem> result;

        if (g_warn_queue.empty())
            return result;
        if (limit <= 0)
            limit = 100;

        if (after_seq_id == 0)
        {
            // 模式 1：快照模式 (从队尾往前取最新的 N 条，解决定格 Bug)
            for (auto it = g_warn_queue.rbegin(); it != g_warn_queue.rend(); ++it)
            {
                result.push_back(*it);
                if ((int)result.size() >= limit)
                    break;
            }
            std::reverse(result.begin(), result.end()); // 翻转成时间正序
        }
        else
        {
            // 模式 2：增量模式 (取大于 after_seq_id 的新记录)
            for (auto it = g_warn_queue.begin(); it != g_warn_queue.end(); ++it)
            {
                if (it->seq_id > after_seq_id)
                {
                    result.push_back(*it);
                    if (limit > 0 && (int)result.size() >= limit)
                        break;
                }
            }
        }
        return result;
    }

    // 快速投递函数 (对齐原版：只过滤出开关 CB 与遥信 YX，其他类型不发)
    static bool SendDnetUnflash(const std::vector<int> &key_ids, const std::string &username)
    {
        (void)username;
        if (key_ids.empty())
            return true;

        std::vector<int> valid_ids;
        for (int id : key_ids)
        {
            int table_no = id / 1000000;
            // 严格对齐原版：只允许开关(CB)、遥信(YX)、保护(OTHER)进入清闪流程
            if (table_no == CB_DEVICE_NO || table_no == YX_DEFINE_NO || table_no == OTHER_DEVICE_NO)
            {
                valid_ids.push_back(id);
            }
        }

        if (valid_ids.empty())
            return true;

        // 投递进队列即刻返回，耗时 0.01 毫秒
        {
            std::lock_guard<std::mutex> lk(g_unflash_queue_mutex);
            for (int id : valid_ids)
            {
                g_unflash_point_queue.push_back(id);
            }
        }
        g_unflash_cv.notify_one();
        return true;
    }

    // =============================================================
    // 3. 告警确认接口（单条、批量、全部）
    // =============================================================
    bool AckWarn(uint64_t seq_id, const std::string &username)
    {
        int target_key_id = 0;
        {
            // 内存状态瞬间置为已确认，前端刷新立竿见影
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : g_warn_queue)
            {
                if (item.seq_id == seq_id)
                {
                    item.confirm = 1;
                    item.confirm_desc = "已确认";
                    target_key_id = item.key_id;
                    break;
                }
            }
        }
        if (target_key_id > 0)
        {
            return SendDnetUnflash({target_key_id}, username);
        }
        return true;
    }

    int AckWarnBatch(const std::vector<uint64_t> &seq_ids, const std::string &username)
    {
        std::vector<int> target_key_ids;
        {
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (uint64_t sid : seq_ids)
            {
                for (auto &item : g_warn_queue)
                {
                    if (item.seq_id == sid && item.confirm == 0)
                    {
                        item.confirm = 1;
                        item.confirm_desc = "已确认";
                        if (item.key_id > 0)
                            target_key_ids.push_back(item.key_id);
                        break;
                    }
                }
            }
        }
        if (!target_key_ids.empty())
        {
            SendDnetUnflash(target_key_ids, username);
        }
        return (int)target_key_ids.size();
    }

    int AckAllWarns(const std::string &username)
    {
        std::vector<int> ids;
        {
            // 原版 ack_all 核心逻辑：直接把队列中所有未确认的项置为已确认
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : g_warn_queue)
            {
                if (item.confirm == 0)
                {
                    item.confirm = 1;
                    item.confirm_desc = "已确认";
                    if (item.key_id > 0)
                        ids.push_back(item.key_id);
                }
            }
        }
        if (!ids.empty())
        {
            SendDnetUnflash(ids, username);
        }
        return (int)ids.size();
    }

    // =============================================================
    // ★★★ 原版南瑞 SCADA 告警等级研判算法 (复刻 CWarnLevMap.h) ★★★
    // =============================================================
    static int JudgeScadaWarnLevel(int table_no, int kid, int status)
    {
        // 规则 1：遥信事故分合闸强行置为 3 (事故信息)
        if (table_no == 43 || table_no == 55)
        {
            if (status == 9 || status == 10)
                return 3;
        }

        // 规则 2：查共享内存点表配置的 int_5 (第一优先级点级配置)
        if (kid > 0)
        {
            int tno = kid / 1000000;
            if (tno == YX_DEFINE_NO)
            {
                yx_define_scada yx;
                if (LdbGetRecordById(YX_DEFINE_NO, (char *)&kid, (char *)&yx, sizeof(yx)) > 0)
                {
                    // int_5 对应告警等级 (1:提示, 2:告警, 3:事故)
                    int warntype_id = yx.int_5 % 1000000;
                    if (warntype_id >= 1 && warntype_id <= 3)
                        return warntype_id;
                }
            }
            else if (tno == YC_DEFINE_NO)
            {
                yc_define_scada yc;
                if (LdbGetRecordById(YC_DEFINE_NO, (char *)&kid, (char *)&yc, sizeof(yc)) > 0)
                {
                    int warntype_id = yc.int_5 % 1000000;
                    if (warntype_id >= 1 && warntype_id <= 3)
                        return warntype_id;
                }
            }
        }

        // 规则 3：表级默认兜底 (第二优先级)
        switch (table_no)
        {
        case 44:
            return 2; // 遥测越限/波动异常 -> 告警信息
        case 362:
            return 2; // 微机保护动作 -> 告警信息
        case 364:
            return 2; // 保护自检异常 -> 告警信息
        case 308:
            return 3; // 故障推理结果 -> 事故信息
        case 43:
            return 1; // 正常开关遥信变位 -> 提示信息
        case 55:
            return 1; // 正常SOE -> 提示信息
        default:
            return 1; // 人工遥控、遥调、挂牌等 -> 提示信息
        }
    }

    static std::string gbk_to_utf8(const char *src)
    {
        if (!src || src[0] == '\0')
            return std::string();
        iconv_t cd = iconv_open("UTF-8//IGNORE", "GB18030");
        if (cd == (iconv_t)-1)
        {
            cd = iconv_open("UTF-8//IGNORE", "GBK");
            if (cd == (iconv_t)-1)
                return std::string(src);
        }
        size_t inbytesleft = strlen(src);
        size_t outbytesleft = inbytesleft * 4 + 1;
        std::string out;
        out.resize(outbytesleft);
        char *inbuf = const_cast<char *>(src);
        char *outbuf = &out[0];
        char *outbuf_start = outbuf;
        if (iconv(cd, &inbuf, &inbytesleft, &outbuf, &outbytesleft) == (size_t)-1)
        {
            iconv_close(cd);
            return std::string(src);
        }
        iconv_close(cd);
        out.resize(outbuf - outbuf_start);
        return out;
    }

    static std::string json_escape(const std::string &s)
    {
        std::string out;
        out.reserve(s.size() + 8);
        for (size_t i = 0; i < s.size(); ++i)
        {
            unsigned char c = (unsigned char)s[i];
            switch (c)
            {
            case '"':
                out += "\\\"";
                break;
            case '\\':
                out += "\\\\";
                break;
            case '\b':
                out += "\\b";
                break;
            case '\f':
                out += "\\f";
                break;
            case '\n':
                out += "\\n";
                break;
            case '\r':
                out += "\\r";
                break;
            case '\t':
                out += "\\t";
                break;
            default:
                if (c < 0x20)
                {
                    char buf[8];
                    sprintf(buf, "\\u%04x", c);
                    out += buf;
                }
                else
                    out += (char)c;
            }
        }
        return out;
    }

    class JsonWriter
    {
    public:
        JsonWriter(bool pretty = false, int indent = 2)
            : m_pretty(pretty), m_indent(indent), m_level(0), m_firstInContainer(true), m_afterKey(false) {}
        void beginObject()
        {
            preValue();
            m_ss += '{';
            m_level++;
            m_firstInContainer = true;
        }
        void endObject()
        {
            m_level--;
            newlineIndent();
            m_ss += '}';
            m_firstInContainer = false;
        }
        void beginArray()
        {
            preValue();
            m_ss += '[';
            m_level++;
            m_firstInContainer = true;
        }
        void endArray()
        {
            m_level--;
            newlineIndent();
            m_ss += ']';
            m_firstInContainer = false;
        }
        void key(const std::string &k)
        {
            if (m_firstInContainer)
                m_firstInContainer = false;
            else
            {
                m_ss += ',';
                newlineIndent();
            }
            m_ss += '"';
            m_ss += json_escape(k);
            m_ss += "\":";
            if (m_pretty)
                m_ss += ' ';
            m_afterKey = true;
        }
        void valueString(const std::string &v)
        {
            preValue();
            m_ss += '"';
            m_ss += json_escape(v);
            m_ss += '"';
        }
        void valueInt(int v)
        {
            preValue();
            char buf[32];
            sprintf(buf, "%d", v);
            m_ss += buf;
        }
        void valueInt(long long v)
        {
            preValue();
            char buf[32];
            sprintf(buf, "%lld", v);
            m_ss += buf;
        }
        void valueDouble(double v)
        {
            preValue();
            char buf[64];
            if (v == (long long)v && v < 1e15 && v > -1e15)
                sprintf(buf, "%.0f", v);
            else
                sprintf(buf, "%.4f", v);
            m_ss += buf;
        }
        void valueBool(bool v)
        {
            preValue();
            m_ss += v ? "true" : "false";
        }
        void rawKV(const std::string &k, const std::string &v)
        {
            key(k);
            valueString(v);
        }
        void rawKV(const std::string &k, const char *v)
        {
            key(k);
            valueString(v ? v : "");
        }
        void rawKV(const std::string &k, int v)
        {
            key(k);
            valueInt(v);
        }
        void rawKV(const std::string &k, long long v)
        {
            key(k);
            valueInt(v);
        }
        void rawKV(const std::string &k, double v)
        {
            key(k);
            valueDouble(v);
        }
        void rawKV(const std::string &k, bool v)
        {
            key(k);
            valueBool(v);
        }
        std::string str() const { return m_ss; }

    private:
        void preValue()
        {
            if (m_afterKey)
                m_afterKey = false;
            else
            {
                if (m_firstInContainer)
                    m_firstInContainer = false;
                else
                {
                    m_ss += ',';
                    newlineIndent();
                }
            }
        }
        void newlineIndent()
        {
            if (!m_pretty)
                return;
            m_ss += '\n';
            m_ss.append(m_level * m_indent, ' ');
        }
        bool m_pretty;
        int m_indent;
        int m_level;
        std::string m_ss;
        bool m_firstInContainer;
        bool m_afterKey;
    };

    static int SendYkYtByDnet(system_net_info *sysnet, const char *hostname,
                              int report_type, int dev_id, int yk_id,
                              int up_or_down, unsigned char yk_state,
                              unsigned char channel_no,
                              float set_value, bool is_yt)
    {
        if (!sysnet)
            return -1;
        int tmp_i = 0;
        char tmp_c = 0;
        int number = 1;
        int report_len = is_yt ? 21 : 17;

        CPackStream PackBuf;
        PackBuf.OpenForWrite();
        PackBuf << report_type << tmp_i << tmp_i << tmp_i << tmp_i;
        PackBuf << number << report_len << tmp_c << tmp_c << tmp_c << tmp_c;
        PackBuf << dev_id << yk_id << up_or_down << yk_state << channel_no;
        if (is_yt)
        {
            PackBuf << set_value;
        }

        char *buffer = NULL;
        int counter = 0;
        PackBuf.GetWriteBuffer(buffer, counter);

        int ret = sysnet->dnet_send_message(
            DNET_REPORT_POINT_TO_POINT,
            report_type,
            (char *)hostname,
            (char *)"FRONT_NET",
            buffer, counter);

        free(buffer);
        return ret;
    }

        bool GetYkPointInfoById(int yk_id, YkPointInfo &out)
        {
            yk_yx_rely_scada yk;
            if (LdbGetRecordById(YK_YX_RELY_NO, (char *)&yk_id, (char *)&yk, sizeof(yk)) <= 0)
                return false;
            out.yk_id = yk.yk_id;
            out.yk_type = yk.yk_type;
            out.channel = yk.current_channel;
            out.valid = true;
            strncpy(out.yk_name, yk.yk_name, sizeof(out.yk_name) - 1);
            strncpy(out.yk_alias, yk.yk_alias, sizeof(out.yk_alias) - 1);

            for (auto &c : m_ykChannels)
            {
                if (c.yk_id != yk_id)
                    continue;
                out.gin = c.gin;
                out.order = c.order_no;
                out.channel = c.channel;
                out.exec_type = c.exec_type;
                break;
            }
            out.dev_id = m_dbmap.GetDevIdxByIdx(yk_id);
            return true;
        }

        // ★★★ 按需初始化 DNET 发送，发完即毁，彻底杜绝消息总线堵塞 ★★★
        int DoYk(int yk_id, const char *action, int yk_state)
        {
            YkPointInfo info;
            if (!GetYkPointInfoById(yk_id, info))
                return -1;
            if (!is_yk_type(info.yk_type))
                return -1;
            if (info.dev_id <= 0)
                return -1;

            int report_type = 0;
            int status = 0;
            if (strcmp(action, "prev") == 0)
            {
                report_type = DNET_YK_PRE_SET;
                status = 1;
            }
            else if (strcmp(action, "exec") == 0)
            {
                report_type = DNET_YK_EXECUTE;
                status = 4;
            }
            else if (strcmp(action, "cancel") == 0)
            {
                report_type = DNET_YK_CANCEL;
                status = 5;
            }
            else if (strcmp(action, "direct") == 0)
            {
                report_type = DNET_YK_DIRECT;
                status = 4;
            }
            else
                return -1;

            char local_host[64] = {0};
            ns_gethostname(local_host, sizeof(local_host));
            system_net_info sysnet((char *)"scada_dump", DNET_NO);

            int ret = SendYkYtByDnet(&sysnet, local_host, report_type, info.dev_id, yk_id, 0, (unsigned char)yk_state,
                                     (unsigned char)info.channel, 0.0f, false);
            if (ret < 0)
                return ret;

            SendOperateYkYt(&sysnet, local_host, false, info.dev_id, yk_id, status, yk_state, 0, 0.0f, 0.0f, "scada_dump");
            return ret;
        }

        int DoYt(int yt_id, const char *action, const char *oper, float set_value, float old_value = -1.0f)
        {
            YkPointInfo info;
            if (!GetYkPointInfoById(yt_id, info))
                return -1;
            if (!is_yt_type(info.yk_type))
                return -1;
            if (info.dev_id <= 0)
                return -1;

            int report_type = 0;
            int status = 0;
            if (strcmp(action, "prev") == 0)
            {
                report_type = DNET_YT_PRE_SET;
                status = 1;
            }
            else if (strcmp(action, "exec") == 0)
            {
                report_type = DNET_YT_EXECUTE;
                status = 4;
            }
            else if (strcmp(action, "cancel") == 0)
            {
                report_type = DNET_YT_CANCEL;
                status = 5;
            }
            else if (strcmp(action, "direct") == 0)
            {
                report_type = DNET_YT_DIRECT;
                status = 4;
            }
            else
                return -1;

            int up_or_down = 0;
            unsigned char yk_state = 0;
            if (strcmp(oper, "up") == 0)
            {
                up_or_down = 1;
                yk_state = YT_STATE_UP;
            }
            else if (strcmp(oper, "down") == 0)
            {
                up_or_down = 2;
                yk_state = YT_STATE_DOWN;
            }
            else if (strcmp(oper, "stop") == 0)
            {
                up_or_down = 3;
                yk_state = YT_STATE_STOP;
            }
            else if (strcmp(oper, "adjust") == 0)
            {
                up_or_down = 0;
                yk_state = YT_STATE_ADJUST;
            }
            else
                return -1;

            if (old_value < 0.0f)
                old_value = 0.0f;

            char local_host[64] = {0};
            ns_gethostname(local_host, sizeof(local_host));
            system_net_info sysnet((char *)"scada_dump", DNET_NO);

            int ret = SendYkYtByDnet(&sysnet, local_host, report_type, info.dev_id, yt_id, up_or_down, yk_state,
                                     (unsigned char)info.channel, set_value, true);
            if (ret < 0)
                return ret;

            SendOperateYkYt(&sysnet, local_host, true, info.dev_id, yt_id, status, yk_state, up_or_down, old_value, set_value, "scada_dump");
            return ret;
        }

        std::vector<YkFindItem> FindYkByName(const std::string &keyword, bool yt_only)
        {
            std::vector<YkFindItem> res;
            for (auto &y : m_yks)
            {
                bool isYt = is_yt_type(y.yk_type);
                if (yt_only && !isYt)
                    continue;
                std::string name = gbk_to_utf8(y.yk_name);
                std::string alias = gbk_to_utf8(y.yk_alias);
                if (!keyword.empty() && name.find(keyword) == std::string::npos && alias.find(keyword) == std::string::npos)
                    continue;
                res.push_back({y.yk_id, name, alias, (int)y.yk_type, yk_type_name(y.yk_type), isYt});
            }
            return res;
        }

    private:
        template <typename T, typename F>
        static const T *FindByIdGeneric(const std::vector<T> &v, F fn, int id)
        {
            for (auto &x : v)
                if (fn(x) == id)
                    return &x;
            return nullptr;
        }
        static const fac_info_scada *FindById(const std::vector<fac_info_scada> &v, int id)
        {
            return FindByIdGeneric(v, [](const fac_info_scada &x)
                                   { return x.fac_id; }, id);
        }
        static const bay_info_scada *FindById(const std::vector<bay_info_scada> &v, int id)
        {
            return FindByIdGeneric(v, [](const bay_info_scada &x)
                                   { return x.bay_id; }, id);
        }
        static const cb_device_scada *FindById(const std::vector<cb_device_scada> &v, int id)

    RealtimeResult QueryRealtime(const std::vector<int> &yc_ids,
                                 const std::vector<int> &yx_ids,
                                 const std::vector<int> &dd_ids)
    {
        RealtimeResult res;
        res.yc_list.reserve(yc_ids.size());
        res.yx_list.reserve(yx_ids.size());
        res.dd_list.reserve(dd_ids.size());

        // 1. 遥测直读 (微秒级直查共享内存)
        for (int ycId : yc_ids)
        {
            yc_define_scada y;
            if (LdbGetRecordById(YC_DEFINE_NO, (char *)&ycId, (char *)&y, sizeof(y)) > 0)
            {
                res.yc_list.push_back({ycId, (double)y.yc_val, (int)y.status});
            }
        }

        // 2. 遥信直读 (微秒级直查共享内存)
        for (int yxId : yx_ids)
        {
            yx_define_scada y;
            if (LdbGetRecordById(YX_DEFINE_NO, (char *)&yxId, (char *)&y, sizeof(y)) > 0)
            {
                res.yx_list.push_back({yxId, (int)y.yx_value, (int)y.status, (int)y.q});
            }
        }

        // 3. 电度直读 (微秒级直查共享内存，直接读取 kwh_2)
        for (int ddId : dd_ids)
        {
            dd_define_scada d;
            if (LdbGetRecordById(DD_DEFINE_NO, (char *)&ddId, (char *)&d, sizeof(d)) > 0)
            {
                res.dd_list.push_back({ddId, (double)d.kwh_2, (int)d.status});
            }
        }

        return res;
    }

    // ★★★ 遥信清闪：通过官方实时库接口直接写入 Linux 共享内存 ★★★
    // =============================================================
    UnflashResult ClearYxFlash(const std::vector<int> &yx_ids, bool all)
    {
        int cleared_count = 0;
        CDbMap dbmap;

        // 打开遥信表与一次设备表（整个清闪过程只打开一次，开销极小）
        TableOp yx_table, cb_table;
        if (yx_table.TableOpen(SCADA, YX_DEFINE_NO) < 0)
        {
            return {false, 0, "Failed to open YX_DEFINE table"};
        }
        cb_table.TableOpen(SCADA, CB_DEVICE_NO);

        // 核心清闪逻辑：修改后直接通过 TableUpdate 刷入共享内存
        auto do_clear_one = [&](int id)
        {
            yx_define_scada y;
            // 1. 通过官方直读接口读取当前测点
            if (LdbGetRecordById(YX_DEFINE_NO, (char *)&id, (char *)&y, sizeof(y)) > 0)
            {
                // 检查是否处于闪烁态 (7:变位合, 8:变位分, 9:事故合, 10:事故分, 11:SOE动, 12:SOE复)
                if (y.status >= STATE_YX_BW_ON && y.status <= STATE_YX_SOE_OFF)
                {
                    // ★★★ 1. 遥信测点状态复位为 0 ★★★
                    y.status = STATE_YX_NORMAL;

                    // 落实写入 Linux 共享内存实时库
                    if (yx_table.TableUpdate((char *)&y, sizeof(y)) >= 0)
                    {
                        cleared_count++;
                    }

                    // ★★★ 2. 同步将所属开关本体（cb_device）的共享内存状态复位为 0 ★★★
                    int dev_id = dbmap.GetDevIdxByIdx(id);
                    if (dev_id > 0)
                    {
                        cb_device_scada cb;
                        if (LdbGetRecordById(CB_DEVICE_NO, (char *)&dev_id, (char *)&cb, sizeof(cb)) > 0)
                        {
                            if (cb.status >= STATE_YX_BW_ON && cb.status <= STATE_YX_SOE_OFF)
                            {
                                cb.status = STATE_YX_NORMAL;                   // 开关本体也停止闪烁
                                cb_table.TableUpdate((char *)&cb, sizeof(cb)); // 写入共享内存
                            }
                        }
                    }
                }
            }
        };

        if (all)
        {
            // 【全站全部清闪】：遍历全站遥信表
            auto all_yxs = fetch_table<yx_define_scada>(SCADA, YX_DEFINE_NO);
            for (const auto &yx : all_yxs)
            {
                do_clear_one(yx.yx_id);
            }
        }
        else
        {
            // 【当前画面清闪】：精准清闪传入的测点
            for (int id : yx_ids)
            {
                do_clear_one(id);
            }
        }

        yx_table.TableClose();
        cb_table.TableClose();

        printf("[SCADA SHM] ClearYxFlash: %d points cleared in Shared Memory.\n", cleared_count);
        return {true, cleared_count, "Successfully cleared in shared memory"};
    }

    static void DoRealDnetUnflash(const std::vector<int> &key_ids)
    {
        if (key_ids.empty() || !g_pGlobalDnet)
            return;

        char local_host[32] = {0};
        ns_gethostname(local_host, 32);

        time_t tmp_time = time(NULL);
        int int_time = (int)tmp_time;
        short field_id = 0;
        short status_field = 0;
        int count = (int)key_ids.size();
        int report_type = DNET_YX_UNFLASH;

        CPackStream PackBuf;
        PackBuf.OpenForWrite();
        PackBuf << report_type << int_time << field_id << status_field;
        PackBuf.set_char((char *)local_host, 32);
        PackBuf.set_char((char *)"web_user", 32);
        PackBuf << count;

        for (int id : key_ids)
        {
            PackBuf << id;
        }

        char *buffer = NULL;
        int counter = 0;
        PackBuf.GetWriteBuffer(buffer, counter);

        // ★★★ 对齐原版：独立发送互斥，优先点对点发送主服务器，不发全网慢速等待广播 ★★★
        {
            std::lock_guard<std::mutex> lk(g_dnet_mutex);
            int ret = g_pGlobalDnet->dnet_send_message(
                DNET_REPORT_POINT_TO_POINT,
                DNET_YX_UNFLASH,
                DNET_SERVER_HOST,
                (char *)"svc_scada_server",
                buffer, counter);

            if (ret < 0)
            {
                g_pGlobalDnet->dnet_send_message(
                    DNET_REPORT_POINT_TO_POINT,
                    DNET_YX_UNFLASH,
                    DNET_LOCAL_HOST,
                    (char *)"svc_scada_proxy",
                    buffer, counter);
            }
        }

        if (buffer)
        {
            free(buffer);
            buffer = NULL;
        }
    }

    // 后台独立异步清闪处理线程
    static void UnflashWorker()
    {
        while (g_unflash_running.load())
        {
            std::vector<int> batch;
            {
                std::unique_lock<std::mutex> lk(g_unflash_queue_mutex);
                g_unflash_cv.wait_for(lk, std::chrono::milliseconds(50), []
                                      { return !g_unflash_point_queue.empty() || !g_unflash_running.load(); });

                if (!g_unflash_running.load())
                    break;

                while (!g_unflash_point_queue.empty())
                {
                    batch.push_back(g_unflash_point_queue.front());
                    g_unflash_point_queue.pop_front();
                }
            }

            if (!batch.empty())
            {
                // 测点全局去重，避免重复发送相同的开关/遥信 ID
                std::sort(batch.begin(), batch.end());
                batch.erase(std::unique(batch.begin(), batch.end()), batch.end());

                // 分批发送，每包最多 500 个点
                const size_t CHUNK_SIZE = 500;
                for (size_t i = 0; i < batch.size(); i += CHUNK_SIZE)
                {
                    size_t end = std::min(i + CHUNK_SIZE, batch.size());
                    std::vector<int> chunk(batch.begin() + i, batch.begin() + end);
                    DoRealDnetUnflash(chunk);
                    usleep(5 * 1000); // 微量释放
                }
            }
        }
    }

    // 毫秒级极速包解析与精准分类研判
    // =============================================================
    static void ParseFullWarnPacket(const char *buffer, int length, int byte_order)
    {
        if (!buffer || length <= 0)
            return;

        CPackStream pack;
        pack.OpenForRead((char *)buffer, length, byte_order);

        int num = 0;
        pack >> num;
        if (num <= 0 || num > 1000)
            return;

        CDbMap dbmap;
        time_t now = time(NULL);
        std::vector<WarnItem> parsed_batch;
        parsed_batch.reserve(num);

        static std::once_flag init_level_flag;
        std::call_once(init_level_flag, InitWarnLevelsFromDb);

        for (int i = 0; i < num; ++i)
        {
            WarnItem item;
            item.confirm = 0;
            item.confirm_desc = "未确认";
            item.timestamp = (uint64_t)now;
            item.time_str = FormatTimestamp(now, 0);

            pack >> item.type; // 原始类型

            // 1. 专业大类精确研判 (严格对应截图标签)
            if (item.type == 1)
                item.type_name = "遥信变位";
            else if (item.type == 2 || item.type == 22 || item.type == 23)
                item.type_name = "遥测越限";
            else if (item.type == 3)
                item.type_name = "厂站工况";
            else if (item.type == 4)
                item.type_name = "结点工况";
            else if (item.type == 8 || item.type == 9)
                item.type_name = "遥控";
            else if (item.type == 11 || item.type == 12)
                item.type_name = "人工操作";
            else if (item.type == 10)
                item.type_name = "事件SOE";
            else if (item.type == 28 || item.type == 35 || item.type == 36)
                item.type_name = "保护事件";
            else
                item.type_name = "遥信变位";

            item.my_type = (item.type >= 0 && item.type < 50) ? g_ShowArray[item.type] : 0;
            item.level = (item.type >= 0 && item.type < 50) ? g_ShowArray2[item.type] : 1;

            int warn_no = 0;
            pack >> warn_no;
            bool has_detail = ((warn_no & 0x8000) != 0 && warn_no != -1);

            int str_len = 0;
            pack >> str_len;
            if (str_len > 0 && str_len < 1024)
            {
                std::vector<char> raw_str(str_len + 1, 0);
                pack.get_char(str_len, raw_str.data());
                item.content = gbk_to_utf8(raw_str.data());
            }

            pack >> item.fac_id;
            pack >> item.key_id;
            pack >> item.field_id;
            pack >> item.value;
            item.val_desc = GetValueMeaning(item.type, item.value);

            if (has_detail)
            {
                char result_buf[65] = {0}, host_buf[33] = {0}, oper_buf[33] = {0};
                pack.get_char(64, result_buf);
                pack.get_char(32, host_buf);
                pack.get_char(32, oper_buf);
                std::string res_u = gbk_to_utf8(result_buf);
                std::string op_u = gbk_to_utf8(oper_buf);
                if (!res_u.empty())
                    item.content += " [" + res_u + "]";
                if (!op_u.empty())
                    item.content += " [操作人:" + op_u + "]";
            }

            // 2. 查库内测点属性精准研判严重级别 (原版 main_view_warn.cpp 规则)
            int table_no = item.key_id / 1000000;
            if (table_no == YX_DEFINE_NO)
            {
                yx_define_scada yx_stru;
                if (LdbGetRecordById(YX_DEFINE_NO, (char *)&item.key_id, (char *)&yx_stru, sizeof(yx_define_scada)) > 0)
                {
                    for (int k = 0; k < g_warn_level_count; ++k)
                    {
                        if (g_LevelDefines[k].warntype_id == yx_stru.int_5)
                        {
                            item.level = g_LevelDefines[k].warn_level;
                            break;
                        }
                    }
                }
                // 事故分闸/跳闸特例研判
                if (item.content.find("事故") != std::string::npos &&
                    (item.content.find("跳闸") != std::string::npos || item.content.find("分闸") != std::string::npos))
                {
                    item.level = g_sg_level;
                }
            }
            else if (table_no == YC_DEFINE_NO)
            {
                item.level = 2; // 越限默认归为告警
            }
            else if (table_no == CB_DEVICE_NO)
            {
                if (item.content.find("事故跳闸") != std::string::npos)
                {
                    item.level = g_sg_level;
                }
            }

            // 匹配级别中文与颜色
            if (item.level == 3)
            {
                item.level_name = "事故信息";
                item.level_color = "#FF0000";
            }
            else if (item.level == 2)
            {
                item.level_name = "告警信息";
                item.level_color = "#E6A23C";
            }
            else
            {
                item.level_name = "提示信息";
                item.level_color = "#67C23A";
            }

            // 3. 研判业务态：检修 与 复归状态
            item.is_examine = (item.content.find("检修") != std::string::npos);
            item.is_revert = (item.value == 0 || item.content.find("复归") != std::string::npos);

            // 4. 组装归属标签集合 (前端可直接使用 tabs.includes(当前Tab))
            item.tabs.push_back("时序信息");
            item.tabs.push_back(item.level_name); // 提示信息 / 告警信息 / 事故信息
            item.tabs.push_back(item.type_name);  // 遥信变位 / 遥测越限 / 厂站工况 ...
            if (item.is_examine)
            {
                item.tabs.push_back("检修信息");
            }
            if (!item.is_revert && !item.is_examine)
            {
                item.tabs.push_back("未复归信息");
            }

            // 5. 拼装 SCADA 原始单行全文本
            item.full_text = item.level_name + "  " + item.time_str + "  " + item.content;

            item.status_desc = GetStatusDesc(item.status, item.value);

            // 6. 翻译名称
            if (item.key_id > 0)
            {
                item.fac_id = dbmap.GetFacIdxByIdx(item.key_id);
                item.bay_id = dbmap.GetBayIdxByIdx(item.key_id);
                item.dev_id = dbmap.GetDevIdxByIdx(item.key_id);

                item.fac_name = GetCachedNameByIdx(dbmap, item.fac_id);
                item.bay_name = GetCachedNameByIdx(dbmap, item.bay_id);
                item.dev_name = GetCachedNameByIdx(dbmap, item.dev_id);
                item.key_name = GetCachedNameByIdx(dbmap, item.key_id);
            }
            else
            {
                item.fac_name = "全站系统";
                item.bay_name = "-";
                item.dev_name = "-";
                item.key_name = "-";
            }

            parsed_batch.push_back(item);
        }

        // 推入内存队列
        {
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : parsed_batch)
            {
                item.seq_id = ++g_warn_seq_counter;
                if (g_warn_queue.size() >= MAX_WARN_CACHE)
                {
                    g_warn_queue.pop_front();
                }
                g_warn_queue.push_back(item);
            }
        }
    }

    // =============================================================
    // 高吞吐后台监听线程 (固定使用白名单 "GJ_SERVER")
    // =============================================================

    static void WarnListenWorker()
    {
        // 确保全局唯一的 DNET 句柄已初始化
        if (!g_pGlobalDnet)
        {
            g_pGlobalDnet = new system_net_info((char *)"GJ_SERVER", DNET_NO);
        }

        const int RECV_BUF_SIZE = 65536; // 64KB 大缓冲区
        char *recv_buffer = (char *)malloc(RECV_BUF_SIZE);

        while (g_warn_running.load())
        {
            app_buf_head ahead;
            int recv_count = 0;

            // ★★★ 核心优化：去掉所有互斥锁！dnet_receive_message 独占高速收取，绝不被发送端阻塞 ★★★
            while (true)
            {
                int ret = -1;
                if (g_pGlobalDnet)
                {
                    ret = g_pGlobalDnet->dnet_receive_message(ahead, recv_buffer);
                }

                if (ret >= 0)
                {
                    if (ahead.rhead.report_type == DNET_SEND_MESSAGE_TO_WARN)
                    {
                        ParseFullWarnPacket(recv_buffer, ahead.rhead.length, ahead.rhead.byte_order);
                    }
                    recv_count++;
                    if (recv_count > 50)
                        break; // 批量让出调度片
                }
                else
                {
                    break; // 缓冲区已排空
                }
            }

            if (recv_count == 0)
            {
                usleep(20 * 1000); // 无数据时让出 CPU 20ms
            }
        }

        if (recv_buffer)
        {
            free(recv_buffer);
            recv_buffer = nullptr;
        }
    }

    bool StartWarnListener()
    {
        if (g_warn_running.load())
            return true;
        g_warn_running.store(true);
        g_warn_thread = std::thread(WarnListenWorker);
        g_warn_thread.detach();

        // ★ 启动后台异步清闪线程
        g_unflash_running.store(true);
        g_unflash_thread = std::thread(UnflashWorker);
        g_unflash_thread.detach();

        return true;
    }

    void StopWarnListener()
    {
        g_warn_running.store(false);
        g_unflash_running.store(false);
        g_unflash_cv.notify_all();
    }

    // =============================================================
    // 实时告警查询接口 (彻底修复取数反向导致的停滞定格 Bug)
    // =============================================================
    std::vector<WarnItem> GetRealtimeWarns(int limit, uint64_t after_seq_id)
    {
        std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
        std::vector<WarnItem> result;

        if (g_warn_queue.empty())
            return result;
        if (limit <= 0)
            limit = 100;

        if (after_seq_id == 0)
        {
            // 模式 1：快照模式 (从队尾往前取最新的 N 条，解决定格 Bug)
            for (auto it = g_warn_queue.rbegin(); it != g_warn_queue.rend(); ++it)
            {
                result.push_back(*it);
                if ((int)result.size() >= limit)
                    break;
            }
            std::reverse(result.begin(), result.end()); // 翻转成时间正序
        }
        else
        {
            // 模式 2：增量模式 (取大于 after_seq_id 的新记录)
            for (auto it = g_warn_queue.begin(); it != g_warn_queue.end(); ++it)
            {
                if (it->seq_id > after_seq_id)
                {
                    result.push_back(*it);
                    if (limit > 0 && (int)result.size() >= limit)
                        break;
                }
            }
        }
        return result;
    }

    void ClearRealtimeWarnCache()
    {
        std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
        g_warn_queue.clear();
    }

    // 快速投递函数 (对齐原版：只过滤出开关 CB 与遥信 YX，其他类型不发)
    static bool SendDnetUnflash(const std::vector<int> &key_ids, const std::string &username)
    {
        (void)username;
        if (key_ids.empty())
            return true;

        std::vector<int> valid_ids;
        for (int id : key_ids)
        {
            int table_no = id / 1000000;
            // 严格对齐原版：只允许开关(CB)、遥信(YX)、保护(OTHER)进入清闪流程
            if (table_no == CB_DEVICE_NO || table_no == YX_DEFINE_NO || table_no == OTHER_DEVICE_NO)
            {
                valid_ids.push_back(id);
            }
        }

        if (valid_ids.empty())
            return true;

        // 投递进队列即刻返回，耗时 0.01 毫秒
        {
            std::lock_guard<std::mutex> lk(g_unflash_queue_mutex);
            for (int id : valid_ids)
            {
                g_unflash_point_queue.push_back(id);
            }
        }
        g_unflash_cv.notify_one();
        return true;
    }

    // =============================================================
    // 3. 告警确认接口（单条、批量、全部）
    // =============================================================
    bool AckWarn(uint64_t seq_id, const std::string &username)
    {
        int target_key_id = 0;
        {
            // 内存状态瞬间置为已确认，前端刷新立竿见影
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : g_warn_queue)
            {
                if (item.seq_id == seq_id)
                {
                    item.confirm = 1;
                    item.confirm_desc = "已确认";
                    target_key_id = item.key_id;
                    break;
                }
            }
        }
        if (target_key_id > 0)
        {
            return SendDnetUnflash({target_key_id}, username);
        }
        return true;
    }

    int AckWarnBatch(const std::vector<uint64_t> &seq_ids, const std::string &username)
    {
        std::vector<int> target_key_ids;
        {
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (uint64_t sid : seq_ids)
            {
                for (auto &item : g_warn_queue)
                {
                    if (item.seq_id == sid && item.confirm == 0)
                    {
                        item.confirm = 1;
                        item.confirm_desc = "已确认";
                        if (item.key_id > 0)
                            target_key_ids.push_back(item.key_id);
                        break;
                    }
                }
            }
        }
        if (!target_key_ids.empty())
        {
            SendDnetUnflash(target_key_ids, username);
        }
        return (int)target_key_ids.size();
    }

    int AckAllWarns(const std::string &username)
    {
        std::vector<int> ids;
        {
            // 原版 ack_all 核心逻辑：直接把队列中所有未确认的项置为已确认
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (auto &item : g_warn_queue)
            {
                if (item.confirm == 0)
                {
                    item.confirm = 1;
                    item.confirm_desc = "已确认";
                    if (item.key_id > 0)
                        ids.push_back(item.key_id);
                }
            }
        }
        if (!ids.empty())
        {
            SendDnetUnflash(ids, username);
        }
        return (int)ids.size();
    }

    // =============================================================
    // 4. 完整复刻 OnWBSendReport 打包与接收解包
    // =============================================================
    WarnHistoryResult QueryWarnHistory(const WarnHistoryFilter &filter, int timeout_seconds)
    {
        WarnHistoryResult res;
        res.ok = false;
        res.code = 500;
        res.total = 0;

        // 1. 组装 CPackStream 请求报文 (100% 对应 report_wt.cpp 的 OnWBSendReport)
        CPackStream pack;
        pack.OpenForWrite();

        int start_t = (int)filter.start_time;
        int end_t = (int)filter.end_time;
        int search_type = filter.search_type;

        pack << start_t << end_t << search_type;

        // 厂站列表
        unsigned char select_all_fac = filter.select_all_fac ? 1 : 0;
        pack << select_all_fac;
        if (!select_all_fac)
        {
            short num = (short)filter.fac_ids.size();
            pack << num;
            for (int fid : filter.fac_ids)
                pack << fid;
        }

        // 一次体系 vs 二次体系
        if (search_type == 0)
        {
            // 一次设备：间隔 + 设备
            unsigned char select_all_bay = filter.select_all_bay ? 1 : 0;
            pack << select_all_bay;
            if (!select_all_bay)
            {
                short num = (short)filter.bay_ids.size();
                pack << num;
                for (int bid : filter.bay_ids)
                    pack << bid;
            }

            unsigned char select_all_dev = filter.select_all_dev ? 1 : 0;
            pack << select_all_dev;
            if (!select_all_dev)
            {
                short num = (short)filter.dev_ids.size();
                pack << num;
                for (int did : filter.dev_ids)
                    pack << did;
            }
        }
        else
        {
            // 二次设备：IED装置 + 逻辑设备
            unsigned char select_all_ied = filter.select_all_ied ? 1 : 0;
            pack << select_all_ied;
            if (!select_all_ied)
            {
                short num = (short)filter.ied_ids.size();
                pack << num;
                for (int iid : filter.ied_ids)
                    pack << iid;
            }

            unsigned char select_all_ld = filter.select_all_ld ? 1 : 0;
            pack << select_all_ld;
            if (!select_all_ld)
            {
                short num = (short)filter.ld_ids.size();
                pack << num;
                for (int lid : filter.ld_ids)
                    pack << lid;
            }
        }

        // 主机节点列表
        unsigned char select_all_host = filter.select_all_host ? 1 : 0;
        pack << select_all_host;
        if (!select_all_host)
        {
            short num = (short)filter.host_ids.size();
            pack << num;
            for (int hid : filter.host_ids)
                pack << hid;
        }

        // 等级列表
        unsigned char select_all_level = filter.select_all_level ? 1 : 0;
        pack << select_all_level;
        if (!select_all_level)
        {
            short num = (short)filter.level_ids.size();
            pack << num;
            for (int lid : filter.level_ids)
                pack << lid;
        }

        // 关键测点过滤 (KeyID)
        unsigned char key_id_flag = filter.point_keys.empty() ? 1 : 0;
        pack << key_id_flag;
        if (key_id_flag == 0)
        {
            short num = (short)filter.point_keys.size();
            pack << num;
            for (const auto &pk : filter.point_keys)
            {
                pack << pk.key_id << pk.field_id;
            }
        }

        // 告警子表及状态掩码过滤
        std::vector<WarnTableFilterOption> tables = filter.table_options;
        if (tables.empty())
        {
            // 如果前端未指定，默认勾选 19 张全量子表，掩码全开
            int default_tabs[] = {43, 44, 45, 46, 47, 48, 49, 50, 51, 55, 362, 364, 365, 167, 306, 307, 310, 258, 308};
            for (int tno : default_tabs)
                tables.push_back({tno, 0xFFFFFFFF});
        }
        short table_num = (short)tables.size();
        pack << table_num;
        for (const auto &opt : tables)
        {
            int t_no = opt.table_no;
            int t_mask = (int)opt.status_mask;
            pack << t_no << t_mask;
        }

        char *req_buf = NULL;
        int req_len = 0;
        pack.GetWriteBuffer(req_buf, req_len);

        // 2. 选择服务器发起点对点查询报文
        char dest_server[64] = {0};
        system_net_info query_client((char *)"WebWarnQuery", DNET_NO);
        if (query_client.SelectAServer(dest_server, sizeof(dest_server)) < 0)
        {
            if (req_buf)
                free(req_buf);
            res.message = "Cannot find SCADA query server (svc_query)";
            return res;
        }

        int ret = query_client.dnet_send_message(
            DNET_REPORT_POINT_TO_POINT,
            DNET_DB_WARN_BROWSE_REQUEST,
            dest_server,
            (char *)"svc_query",
            req_buf, req_len);

        if (req_buf)
            free(req_buf);
        if (ret < 0)
        {
            res.message = "Failed to send query request to svc_query";
            return res;
        }

        // 3. 阻塞接收并拼接所有分包 (处理分包序号与 end_flag)
        char *recv_buf = (char *)malloc(65536);
        time_t begin_wait = time(NULL);
        bool query_success = false;
        std::vector<char> full_answer;

        while (time(NULL) - begin_wait < timeout_seconds)
        {
            app_buf_head ahead;
            int r = query_client.dnet_receive_message(ahead, recv_buf);
            if (r >= 0 && ahead.rhead.report_type == DNET_DB_WARN_BROWSE_ANSWER)
            {
                if (ahead.rhead.length >= 4)
                {
                    short end_flag = 0;
                    short order_no = 0;
                    memcpy(&end_flag, recv_buf, sizeof(short));
                    memcpy(&order_no, recv_buf + 2, sizeof(short));

                    size_t chunk_size = ahead.rhead.length - 4;
                    size_t old_size = full_answer.size();
                    full_answer.resize(old_size + chunk_size);
                    memcpy(full_answer.data() + old_size, recv_buf + 4, chunk_size);

                    if (end_flag == 1)
                    {
                        query_success = true;
                        break;
                    }
                }
            }
            else
            {
                usleep(30 * 1000);
            }
        }

        if (recv_buf)
            free(recv_buf);

        if (!query_success || full_answer.empty())
        {
            res.message = "Query timeout or no response from svc_query";
            return res;
        }

        // 4. 解析全量历史记录流 (对齐 warn_view_wt.cpp OnInitialUpdate 多态解包)
        int offset = 0;
        unsigned char over_flag = full_answer[offset++];
        unsigned char msec_flag = full_answer[offset++];
        int record_count = 0;
        memcpy(&record_count, full_answer.data() + offset, sizeof(int));
        offset += sizeof(int);

        if (record_count <= 0)
        {
            res.ok = true;
            res.code = 200;
            res.total = 0;
            res.message = "No records found in range";
            return res;
        }

        CDbMap dbmap;
        int parsed_count = 0;

        while (offset + (int)sizeof(int) <= (int)full_answer.size() && parsed_count < record_count)
        {
            if (filter.limit > 0 && (int)res.items.size() >= filter.limit)
                break;

            WarnHistoryItem item;

            // 发生时间
            int rec_time = 0;
            memcpy(&rec_time, full_answer.data() + offset, sizeof(int));
            offset += sizeof(int);
            item.record_time = (uint64_t)rec_time;

            short msec = 0;
            if (msec_flag == 1)
            {
                memcpy(&msec, full_answer.data() + offset, sizeof(short));
                offset += sizeof(short);
            }
            item.msec = (int)msec;
            item.time_str = FormatTimestamp(rec_time, msec);

            // 表号
            int table_no = 0;
            memcpy(&table_no, full_answer.data() + offset, sizeof(int));
            offset += sizeof(int);
            item.table_no = table_no;
            item.table_name = GetTableNameByNo(table_no);

            // 别名
            char alias_buf[65] = {0};
            memcpy(alias_buf, full_answer.data() + offset, 64);
            offset += 64;
            item.raw_alias = alias_buf;

            int key_id = dbmap.GetIdxByAlias(alias_buf);
            item.key_id = key_id;
            item.fac_id = dbmap.GetFacIdxByIdx(key_id);
            item.bay_id = dbmap.GetBayIdxByIdx(key_id);
            item.dev_id = dbmap.GetDevIdxByIdx(key_id);

            char fac_name[128] = {0};
            char obj_name[128] = {0};
            dbmap.GetNameByIdx(item.fac_id, fac_name, sizeof(fac_name));
            dbmap.GetNameByIdx(key_id, obj_name, sizeof(obj_name));

            std::string fac_utf8 = gbk_to_utf8(fac_name);
            std::string obj_utf8 = gbk_to_utf8(obj_name);

            // 状态
            int status = 0;
            memcpy(&status, full_answer.data() + offset, sizeof(int));
            offset += sizeof(int);
            item.status = status;

            // 根据不同的表解析特定字段 (遥测越限值、遥控操作详情、保护描述等)
            if (table_no == 44)
            {
                // 遥测越限: 带 float 越限值
                float yc_val = 0.0f;
                memcpy(&yc_val, full_answer.data() + offset, sizeof(float));
                offset += sizeof(float);
                item.val = yc_val;

                char val_str[32];
                snprintf(val_str, sizeof(val_str), " 越限值: %.2f", yc_val);
                item.content = fac_utf8 + " " + obj_utf8 + val_str;
            }
            else if (table_no == 50 || table_no == 51)
            {
                // 遥控/遥调: 带操作性质、操作机器、操作人
                int mode = 0;
                memcpy(&mode, full_answer.data() + offset, sizeof(int));
                offset += sizeof(int);

                char op_mach[33] = {0};
                char op_user[33] = {0};
                memcpy(op_mach, full_answer.data() + offset, 32);
                offset += 32;
                memcpy(op_user, full_answer.data() + offset, 32);
                offset += 32;

                item.host_name = op_mach;
                item.operator_name = gbk_to_utf8(op_user);
                item.content = fac_utf8 + " " + obj_utf8 + (table_no == 50 ? " 遥控操作" : " 遥调操作") +
                               " [操作人: " + item.operator_name + "]";
            }
            else if (table_no == 362)
            {
                // 保护事件: 带故障号、事件描述
                int event_no = 0;
                memcpy(&event_no, full_answer.data() + offset, sizeof(int));
                offset += sizeof(int);
                int event_state = 0;
                memcpy(&event_state, full_answer.data() + offset, sizeof(int));
                offset += sizeof(int);
                unsigned char has_report = full_answer[offset++];
                int fault_no = 0;
                memcpy(&fault_no, full_answer.data() + offset, sizeof(int));
                offset += sizeof(int);
                int event_id = 0;
                memcpy(&event_id, full_answer.data() + offset, sizeof(int));
                offset += sizeof(int);

                item.content = fac_utf8 + " " + obj_utf8 + " 保护动作事件";
                if (has_report)
                    item.content += " (有详细报告)";
            }
            else
            {
                // 常规变位与工况事件
                item.content = fac_utf8 + " " + obj_utf8 + " " + GetStatusDesc(status, 1);
            }

            // 文本模糊检索匹配
            if (!filter.keyword.empty() && item.content.find(filter.keyword) == std::string::npos)
            {
                parsed_count++;
                continue;
            }

            res.items.push_back(item);
            parsed_count++;
        }

        res.ok = true;
        res.code = 200;
        res.total = (int)res.items.size();
        res.message = "success";
        return res;
    }
    // =============================================================
    // 5. 智能故障推理报告实现 (复刻 cfuncWarn.cpp repos_xl_gz / repos_mx_gz)
    // =============================================================
    FaultReportResult GetFaultReport(int key_id, uint64_t timestamp)
    {
        FaultReportResult report;
        report.ok = false;

        CDbMap dbmap;
        int dev_id = dbmap.GetDevIdxByIdx(key_id);
        int bay_id = dbmap.GetBayIdxByIdx(key_id);
        int fac_id = dbmap.GetFacIdxByIdx(key_id);

        char fac_name[128] = {0};
        char bay_name[128] = {0};
        dbmap.GetNameByIdx(fac_id, fac_name, sizeof(fac_name));
        dbmap.GetNameByIdx(bay_id, bay_name, sizeof(bay_name));

        report.fac_name = gbk_to_utf8(fac_name);
        report.bay_name = gbk_to_utf8(bay_name);

        // 收集该事件前后 3 秒内本间隔的关联事件
        {
            std::lock_guard<std::mutex> lock(g_warn_queue_mutex);
            for (const auto &w : g_warn_queue)
            {
                if (w.bay_id == bay_id && std::abs((int64_t)w.timestamp - (int64_t)timestamp) <= 3)
                {
                    report.related_events.push_back(w.time_str + " " + w.content);
                }
            }
        }
        report.fault_type_name = "断路器动作 / 保护联动跳闸分析报告";
        report.report_detail = "系统监测到间隔开关跳闸变位，经过时序与拓扑关系综合比对，重合闸及出线保护动作正常。处理建议：检查保护装置定值及一次设备接地情况。";
        report.ok = true;
        return report;
    }
